cmake_minimum_required(VERSION 3.20)
project(ControlFGHDRGuard LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
add_library(ControlFGHDRGuard SHARED src/guard.cpp)
target_include_directories(ControlFGHDRGuard PRIVATE src generated)
target_compile_definitions(ControlFGHDRGuard PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(ControlFGHDRGuard PRIVATE /W4 /WX /guard:cf)
target_link_options(ControlFGHDRGuard PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(ControlFGHDRGuard PRIVATE minhook bcrypt user32 d3d12)
add_executable(policy_test tests/policy_test.cpp)
target_include_directories(policy_test PRIVATE src)
target_compile_options(policy_test PRIVATE /W4 /WX)
add_executable(load_test tests/load_test.cpp)
enable_testing()
add_test(NAME policy COMMAND policy_test)
add_test(NAME helper_loader COMMAND load_test $<TARGET_FILE:ControlFGHDRGuard>)
add_executable(core_load_test tests/core_load_test.cpp)
target_link_libraries(core_load_test PRIVATE minhook)
