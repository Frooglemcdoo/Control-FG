param([string]$Root = (Get-Location).Path)
$ErrorActionPreference = 'Stop'
function Read-Utf8([string]$rel) {
  $p=Join-Path $Root $rel
  if(!(Test-Path -LiteralPath $p)){ throw "GI24 missing required file: $rel" }
  return ([IO.File]::ReadAllText($p,[Text.UTF8Encoding]::new($false))).Replace("`r`n","`n")
}
function Write-Utf8([string]$rel,[string]$text) {
  $p=Join-Path $Root $rel
  [IO.File]::WriteAllText($p,$text,[Text.UTF8Encoding]::new($false))
}
function Replace-Exact([string]$rel,[string]$old,[string]$new) {
  $s=Read-Utf8 $rel
  if($s.Contains($new)){ return }
  if(!$s.Contains($old)){ throw "GI24 patch contract missing in $rel" }
  Write-Utf8 $rel ($s.Replace($old,$new))
}
