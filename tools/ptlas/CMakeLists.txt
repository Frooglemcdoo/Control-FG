cmake_minimum_required(VERSION 3.24)
project(ControlPTLASValidation LANGUAGES CXX)
enable_testing()
add_executable(rt-as-route ../test-rt-as-build-route.cpp)
add_executable(rt-ptlas-plan ../test-rt-ptlas-plan.cpp)
add_executable(rt-as-ownership ../test-rt-as-ownership.cpp)
foreach(target rt-as-route rt-ptlas-plan rt-as-ownership)
 target_compile_features(${target} PRIVATE cxx_std_17)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX /UNDEBUG)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 endif()
 add_test(NAME ${target} COMMAND ${target})
endforeach()
add_executable(rt-capture-replay replay-cpu.cpp)
target_compile_features(rt-capture-replay PRIVATE cxx_std_17)
if(WIN32)
 include(FetchContent)
 FetchContent_Declare(nvapi GIT_REPOSITORY https://github.com/NVIDIA/nvapi.git GIT_TAG 70d337db9186e968eab622f7e786de7e437faf3d)
 FetchContent_MakeAvailable(nvapi)
 find_program(DXC_EXECUTABLE dxc REQUIRED)
 add_executable(Control-PTLAS-Validate validate.cpp)
 target_compile_features(Control-PTLAS-Validate PRIVATE cxx_std_17)
 target_compile_definitions(Control-PTLAS-Validate PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(Control-PTLAS-Validate PRIVATE /W4 /EHsc)
 target_include_directories(Control-PTLAS-Validate PRIVATE ${nvapi_SOURCE_DIR})
 target_link_libraries(Control-PTLAS-Validate PRIVATE d3d12 dxgi dxguid ${nvapi_SOURCE_DIR}/amd64/nvapi64.lib)
 add_custom_command(TARGET Control-PTLAS-Validate POST_BUILD
  COMMAND ${DXC_EXECUTABLE} -T cs_6_5 -E main -O3 -WX ${CMAKE_CURRENT_SOURCE_DIR}/probe.hlsl -Fo $<TARGET_FILE_DIR:Control-PTLAS-Validate>/probe.dxil
  COMMAND ${DXC_EXECUTABLE} -T cs_6_5 -E main -O3 -WX ${CMAKE_CURRENT_SOURCE_DIR}/replay.hlsl -Fo $<TARGET_FILE_DIR:Control-PTLAS-Validate>/replay.dxil
  VERBATIM)
endif()
